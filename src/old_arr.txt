            RResult::RArr(xs) => {
                let mut out = "\{\"ok\":true,\"type\":\"array\",\"value\":["
                let mut first = true
                loop v in xs {
                    if !first { out = out + "," }
                    first = false
                    out = out + "\"" + h.json_escape(v) + "\""
                }
                out = out + "]\}"
                return out
            }
