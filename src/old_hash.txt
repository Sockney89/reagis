            RResult::RHashArr(m) => {
                let ks = m.keys()
                let mut out = "\{\"ok\":true,\"type\":\"hash\",\"value\":\{"
                let mut first = true
                loop k in ks {
                    if !first { out = out + "," }
                    first = false
                    let v = m[k]
                    out = out + "\"" + h.json_escape(k) + "\":\"" + h.json_escape(v) + "\""
                }
                out = out + "\}\}"
                return out
            }
