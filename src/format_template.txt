    fn format(self, r: RResult) -> Str {
        match r {
            RResult::RSimple(s) => { return "+" + s + "\r\n" }
            RResult::RInt(n) => { return ":" + str(n) + "\r\n" }
            RResult::RNil => { return "$-1\r\n" }
            RResult::RBulk(s) => {
                let n = s.length()
                return "$" + str(n) + "\r\n" + s + "\r\n"
            }
            RResult::RArr(xs) => {
                let mut sb = StrBuf { parts: [] }
                sb.append("*")
                sb.append_int(xs.length())
                sb.append("\r\n")
                loop v in xs {
                    sb.append("$")
                    sb.append_int(v.length())
                    sb.append("\r\n")
                    sb.append(v)
                    sb.append("\r\n")
                }
                return sb.to_string()
            }
            RResult::RSetArr(xs) => {
                let mut sb = StrBuf { parts: [] }
                sb.append("*")
                sb.append_int(xs.length())
                sb.append("\r\n")
                loop v in xs {
                    sb.append("$")
                    sb.append_int(v.length())
                    sb.append("\r\n")
                    sb.append(v)
                    sb.append("\r\n")
                }
                return sb.to_string()
            }
            RResult::RHashArr(m) => {
                let ks = m.keys()
                let mut sb = StrBuf { parts: [] }
                sb.append("*")
                sb.append_int(ks.length() * 2)
                sb.append("\r\n")
                loop k in ks {
                    let v = m[k]
                    sb.append("$")
                    sb.append_int(k.length())
                    sb.append("\r\n")
                    sb.append(k)
                    sb.append("\r\n")
                    sb.append("$")
                    sb.append_int(v.length())
                    sb.append("\r\n")
                    sb.append(v)
                    sb.append("\r\n")
                }
                return sb.to_string()
            }
            RResult::RErr(s) => { return "-" + s + "\r\n" }
        }
    }
