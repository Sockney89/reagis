            RResult::RArr(xs) => {
                let mut sb = StrBuf { parts: [] }
                sb.append("\{\"ok\":true,\"type\":\"array\",\"value\":[")
                let mut first = true
                loop v in xs {
                    if !first { sb.append(",") }
                    first = false
                    sb.append("\"")
                    sb.append(h.json_escape(v))
                    sb.append("\"")
                }
                sb.append("]\}")
                return sb.to_string()
            }
